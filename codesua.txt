import java.io.FileReader;
import java.io.FileWriter;
import java.io.IOException;
import java.time.LocalDate;
import java.time.LocalDateTime;
import java.time.format.DateTimeFormatter;
import java.time.format.DateTimeParseException;
import java.util.UUID;
import org.json.simple.JSONArray;
import org.json.simple.JSONObject;
import org.json.simple.parser.JSONParser;
import org.json.simple.parser.ParseException;

public class PersonalTaskManagerViolations {

    private static final String DB_FILE_PATH = "tasks_database.json";
    private static final DateTimeFormatter DATE_FORMATTER = DateTimeFormatter.ofPattern("yyyy-MM-dd");

    // ✅ KHÔNG ĐỔI: Đọc dữ liệu từ file
    private static JSONArray loadTasksFromDb() {
        JSONParser parser = new JSONParser();
        try (FileReader reader = new FileReader(DB_FILE_PATH)) {
            Object obj = parser.parse(reader);
            if (obj instanceof JSONArray) {
                return (JSONArray) obj;
            }
        } catch (IOException | ParseException e) {
            System.err.println("Lỗi khi đọc file database: " + e.getMessage());
        }
        return new JSONArray();
    }

    // ✅ KHÔNG ĐỔI: Ghi dữ liệu xuống file
    private static void saveTasksToDb(JSONArray tasksData) {
        try (FileWriter file = new FileWriter(DB_FILE_PATH)) {
            file.write(tasksData.toJSONString());
            file.flush();
        } catch (IOException e) {
            System.err.println("Lỗi khi ghi vào file database: " + e.getMessage());
        }
    }

    // 📌 ĐÃ THÊM: Hàm kiểm tra mức độ ưu tiên
    private boolean isValidPriority(String priorityLevel) {
        // 🛠 Tách logic kiểm tra ra riêng để tránh lặp lại (áp dụng DRY và SRP)
        String[] validPriorities = {"Thấp", "Trung bình", "Cao"};
        for (String valid : validPriorities) {
            if (valid.equals(priorityLevel)) return true;
        }
        return false;
    }

    // 📌 ĐÃ THÊM: Hàm kiểm tra nhiệm vụ trùng
    private boolean isDuplicate(JSONArray tasks, String title, String dueDateStr) {
        // 🛠 Tách đoạn kiểm tra trùng nhiệm vụ để dễ kiểm thử, dễ tái sử dụng (SRP, DRY)
        for (Object obj : tasks) {
            JSONObject existingTask = (JSONObject) obj;
            if (existingTask.get("title").toString().equalsIgnoreCase(title) &&
                existingTask.get("due_date").toString().equals(dueDateStr)) {
                return true;
            }
        }
        return false;
    }

    // 📌 ĐÃ THÊM: Hàm tạo đối tượng nhiệm vụ
    private JSONObject createTask(String title, String description, String dueDateStr, String priorityLevel, boolean isRecurring) {
        // 🛠 Gom phần tạo JSONObject lại để code dễ đọc, dễ bảo trì (SRP)
        JSONObject newTask = new JSONObject();
        String taskId = UUID.randomUUID().toString(); // vẫn dùng UUID, có thể đổi sang int nếu cần
        String now = LocalDateTime.now().format(DateTimeFormatter.ISO_DATE_TIME);

        newTask.put("id", taskId);
        newTask.put("title", title);
        newTask.put("description", description);
        newTask.put("due_date", dueDateStr);
        newTask.put("priority", priorityLevel);
        newTask.put("status", "Chưa hoàn thành");
        newTask.put("created_at", now);
        newTask.put("last_updated_at", now);
        newTask.put("is_recurring", isRecurring); // ⚠️ Giữ lại is_recurring nhưng bỏ recurrence_pattern (xem tiếp bên dưới)

        // ❌ ĐÃ BỎ: newTask.put("recurrence_pattern", "Chưa xác định");
        // ✅ Lý do: Vi phạm nguyên tắc YAGNI – chưa dùng thì không cần thêm

        return newTask;
    }

    public JSONObject addNewTaskWithViolations(String title, String description,
                                               String dueDateStr, String priorityLevel,
                                               boolean isRecurring) {
        // ✅ Vẫn dùng logic cũ, nhưng đã tách các phần con ra hàm riêng để ngắn gọn

        // Kiểm tra tiêu đề
        if (title == null || title.trim().isEmpty()) {
            System.out.println("Lỗi: Tiêu đề không được để trống.");
            return null;
        }

        // Kiểm tra ngày đến hạn
        LocalDate dueDate;
        if (dueDateStr == null || dueDateStr.trim().isEmpty()) {
            System.out.println("Lỗi: Ngày đến hạn không được để trống.");
            return null;
        }
        try {
            dueDate = LocalDate.parse(dueDateStr, DATE_FORMATTER);
        } catch (DateTimeParseException e) {
            System.out.println("Lỗi: Ngày đến hạn không hợp lệ. Vui lòng sử dụng định dạng YYYY-MM-DD.");
            return null;
        }

        // Kiểm tra mức độ ưu tiên
        if (!isValidPriority(priorityLevel)) {
            System.out.println("Lỗi: Mức độ ưu tiên không hợp lệ. Vui lòng chọn từ: Thấp, Trung bình, Cao.");
            return null;
        }

        // Tải dữ liệu và kiểm tra trùng
        JSONArray tasks = loadTasksFromDb();
        if (isDuplicate(tasks, title, dueDateStr)) {
            System.out.printf("Lỗi: Nhiệm vụ '%s' đã tồn tại với cùng ngày đến hạn.\n", title);
            return null;
        }

        // Tạo và lưu nhiệm vụ
        JSONObject newTask = createTask(title, description, dueDateStr, priorityLevel, isRecurring);
        tasks.add(newTask);
        saveTasksToDb(tasks);

        System.out.println("✅ Đã thêm nhiệm vụ mới thành công với ID: " + newTask.get("id"));
        return newTask;
    }

    public static void main(String[] args) {
        PersonalTaskManagerViolations manager = new PersonalTaskManagerViolations();

        System.out.println("\n1. Thêm nhiệm vụ hợp lệ:");
        manager.addNewTaskWithViolations(
                "Mua sách",
                "Sách Công nghệ phần mềm.",
                "2025-07-20",
                "Cao",
                false
        );

        System.out.println("\n2. Thêm nhiệm vụ trùng lặp:");
        manager.addNewTaskWithViolations(
                "Mua sách",
                "Sách Công nghệ phần mềm.",
                "2025-07-20",
                "Cao",
                false
        );

        System.out.println("\n3. Thêm nhiệm vụ lặp lại:");
        manager.addNewTaskWithViolations(
                "Tập thể dục",
                "Tập gym 1 tiếng.",
                "2025-07-21",
                "Trung bình",
                true
        );

        System.out.println("\n4. Thêm nhiệm vụ không hợp lệ (tiêu đề rỗng):");
        manager.addNewTaskWithViolations(
                "",
                "Không có tiêu đề.",
                "2025-07-22",
                "Thấp",
                false
        );
    }
}
